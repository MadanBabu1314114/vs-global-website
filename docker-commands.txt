# VS GLOBAL — Docker commands (Docker-only run)
# Build the image (run from /workspace/vs-global-react):
docker build -t vs-global-react .
# Run on http://localhost:8080:
docker run -d --name vs-global -p 8080:80 vs-global-react
# Logs / stop / remove:
docker logs -f vs-global
docker stop vs-global
docker rm vs-global
# Rebuild after code changes:
docker stop vs-global; docker rm vs-global; docker build -t vs-global-react .; docker run -d --name vs-global -p 8080:80 vs-global-react
# Dev without Docker (optional):
npm install
npm run dev
